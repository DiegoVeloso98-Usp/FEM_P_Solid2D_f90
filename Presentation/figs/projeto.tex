\begin{frame}

        \framesubtitle{Deformação Não-Linear de Engenharia}

        \begin{figure}[!h]
            \centering
            \includegraphics[width=0.4\textwidth]{figs/deformação.png}
            \scriptsize \caption{ Deformação}
        \end{figure}

        \begin{equation}
            d\vec{y} = \nabla (\vec{f}) \cdot d\vec{x} \quad \Rightarrow \quad   \frac{dy^2}{dx^2} = \vec{u} ^T \cdot A^T  A \cdot \vec{u}
        \end{equation}

        \begin{equation}
            \left.
            \begin{aligned}
                \varepsilon_{uv} &= \frac{1}{2}\left[\frac{\pi}{2}-\cos^{-1}\left(\frac{\vec{u}^T A^T A \vec{v}}{\lambda_u\lambda_v}\right)\right] \\
                \varepsilon_u &= \sqrt{\vec{u}^T A^T A \cdot \vec{u} } - 1  
            \end{aligned}
            \right\}
            \quad \text{Não Objetiva}
        \end{equation}

\end{frame}

\begin{frame}
    \frametitle{Desenvolvimento Teórico}
        \framesubtitle{Deformação Quadrática ou de Green-Lagrange}

        \begin{itemize}
            \item \textbf{Relação entre \(\mathbf{A}\) e \(\vec{u}\):}
        \end{itemize}

        \begin{equation}
            \vec{y} = \vec{x} + \vec{u} \quad \Rightarrow \quad \nabla \vec{y} = \nabla \vec{x} + \nabla \vec{u} \quad \Rightarrow \quad \mathbf{A} = \mathbf{I} + \nabla \vec{u}
        \end{equation}

        \begin{itemize}
            \item \textbf{Deformação quadrática:}
        \end{itemize}

        \begin{equation}
            \frac{dy^2 - dx^2}{dx^2} =  \vec{u}^T \cdot \left(\mathbf{A} ^T \mathbf{A}  - \mathbf{I}  \right) \cdot \vec{u} 
        \end{equation}

        \begin{equation}
            \scalebox{1.0}{$
            \mathbf{C}  := \mathbf{A} ^T \mathbf{A}  \quad \Rightarrow \quad \mathbf{E} = \frac{1}{2} \left(\mathbf{C}  - \mathbf{I} \right)
            $}
        \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Desenvolvimento Teórico}
        \framesubtitle{Equilíbrio Local Euleriano}
        \begin{itemize}
            \item Para satisfazer o equilíbrio de momentos:
        \end{itemize}

        \begin{equation}
            \int\limits_{A} \left(\vec{r} \times \vec{t}\right) da  + \int\limits_{\upsilon} \left(\vec{r} \times \vec{b}\right) d\upsilon  \quad \Rightarrow \quad \underbrace{\sigma_{ij} = \sigma_{ji}}_{\text{Teorema de Cauchy}}
        \end{equation}
        \begin{itemize}
            \item Assim, para definir completamente o equilíbrio :
        \end{itemize}

        \begin{equation}
             \vec{t} = \boldsymbol{\sigma} \cdot \vec{n} \quad \Rightarrow \quad \underbrace{\overbrace{\int\limits_{A} \left(\boldsymbol{\sigma} \cdot \vec{n}\right) da }^{\int\limits_{V} \left[div(\boldsymbol{\sigma})\right] d\upsilon} + \int\limits_{\upsilon} \vec{b}  d\upsilon = 0 }_{\text{Equilíbrio Global}}
        \end{equation}
        % \begin{itemize}
        %     \item Aplica-se o teorema da divergência \(\int\limits_{A} \left(\boldsymbol{\sigma} \cdot \vec{n}\right) da = \int\limits_{V} \left[div(\boldsymbol{\sigma})\right] d\upsilon\)
        % \end{itemize}
        \begin{equation}
            \int\limits_{V} \left[div(\boldsymbol{\sigma})\right] d\upsilon + \int\limits_{\upsilon} \vec{b}  d\upsilon = 0 \quad \Leftrightarrow  \quad \underbrace{\text{div} \boldsymbol{\sigma}  + \vec{b} = 0}_{\text{Equilíbrio Local}}
        \end{equation}


\end{frame}

\begin{frame}
    \frametitle{Desenvolvimento Teórico}
        \framesubtitle{Equilíbrio Local Lagrangiano}
        \begin{itemize}
            \item Partindo do Equilíbrio global Euleriano: \(\int\limits_{A} \left(\boldsymbol{\boldsymbol{\sigma}} \cdot \vec{n}\right) da  + \int\limits_{\upsilon} \vec{b}  d\upsilon = 0\)
        \end{itemize}

        \begin{itemize}
            \item Recorda-se que \(d\upsilon = J d\upsilon_R\) e que \(\vec{n}\cdot da = J \cdot \mathbf{A}^{-T} \vec{n}_R \cdot dA_R\)
        \end{itemize}

        \begin{equation}
            \int\limits_{A_r} \left(\boldsymbol{\sigma} \cdot J \cdot \mathbf{A}^{-T} \vec{n}_R\right) da_R  + \int\limits_{\upsilon_R} \underbrace{\vec{b} J}_{=\vec{b^0}} d\upsilon_R  = 0 \quad \left(\text{Lagrangiano}\right)
        \end{equation}

        \begin{itemize}
            \item 1º tensor de Piola: \(\mathbf{P}= J \mathbf{\boldsymbol{\sigma}}\cdot A^{-T} \quad \left(P_{ij} \neq P_{ji}\right)\):
        \end{itemize}

        \begin{equation}
            \int\limits_{A_R} \left(\mathbf{P} \vec{n}_R\right) da_R  + \int\limits_{\upsilon_R} \vec{b^0} d\upsilon_R  = 0 \quad \left(\text{Forma Fraca}\right)
        \end{equation}

        \begin{equation}
            \mathbf{P} \vec{n}_R + \vec{b^0} d\upsilon_R  = 0 \quad \left(\text{Forma Forte}\right)
        \end{equation}



\end{frame}

\begin{frame}
    \frametitle{Desenvolvimento Teórico}
        \framesubtitle{Estacionariedade da Energia Mecânica Total $\varPi$} 

        \begin{equation}
            \delta \varPi = \int\limits_{\upsilon_0} \left(\mathbf{P} + b^0 -\rho^0 \ddot{y} \right) \delta\vec{Y} d\upsilon_0 = 0
        \end{equation}

        \begin{equation}
            \scalebox{0.9}{$
            \int\limits_{\upsilon_0} \rho^0\ddot{y_i}\delta Y_i d\upsilon_0 - \int\limits_{\upsilon_0} b_i^0\delta Y_i d\upsilon_0 - \int\limits_{A_0} p_i^0\delta Y_i dA_0 - F_i\delta Y_i + \int\limits_{\upsilon_0} P_{ji}\delta y_{i,j}dV_0 = 0
            $}
        \end{equation}
        
        \begin{itemize}
            \item \(\delta Y_{i,j} = \delta A_{ij}\)
        \end{itemize}

                
        \begin{itemize}
            \item Tensor de Tensões de Piola-Kirchhoff de Segunda Espécie : \( P_{ji} = A_{ik}\cdot S_{kj} \) 
        \end{itemize}

        \begin{equation}
            \scalebox{0.85}{$
            \int\limits_{\upsilon_0} \rho_0\ddot{y_i}\delta Y_i d\upsilon_0 - \int\limits_{\upsilon_0} b_i^0\delta Y_i d\upsilon_0 - \int\limits_{A_0} p_i^0\delta Y_i dA_0 - F_i\delta Y_i + \int\limits_{\upsilon_0} \underbrace{A_{ik}\cdot S_{kj}\delta A_{ij}}_{\delta E : S}dV_0 = 0
            $}
        \end{equation}

\end{frame}

\begin{frame}
    \frametitle{Desenvolvimento Teórico}
        \framesubtitle{Estacionariedade da Energia Mecânica Total $\varPi$} 

        
        \begin{itemize}
            \item Finalmente, o equilíbrio Lagrangiano fica definido:
        \end{itemize}

        \begin{equation}
            \scalebox{0.9}{$
            \underbrace{
            \int\limits_{\upsilon_0}\rho^0\ddot{y_i}\delta Y_i d\upsilon_0 
            - \int\limits_{\upsilon_0}b_i^0\delta Y_i d\upsilon_0 
            - \int\limits_{A_0}p_i^0\delta Y_i dA_0 - F_i\delta Y_i
            }_{\text{Forças Externas}}
            + \underbrace{
            \int\limits_{\upsilon_0}S : \delta E d\upsilon_0
            }_{\text{Forças Internas}} = 0
            $}
        \end{equation}

        \begin{itemize}
            \item Conjugados Energéticos: \( P^T = \frac{\partial u_e}{\partial A} \quad \text{e} \quad S = \frac{\partial u_e}{\partial E} \)
        \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Desenvolvimento Teórico}
        \framesubtitle{Energia Específica de Deformação} 

        \begin{itemize}
            \item \textbf{Energia de Deformação Total \(U_e = \int\limits_{\upsilon_0}{u_e \upsilon_0}\):}
            \item \textbf{Modelo Hiperelástico:} É aquele no qual a energia de deformação pode ser equacionada.
            \item \textbf{Modelo Hiperelástico Completo:} 
            \begin{itemize}
                \item \(u_e(C=I)=u_e(E=0)=0\) (Condição de Normalidade)
                \vspace{1cm} 
                \item \(
                    \left.
                    \begin{aligned}
                        u_e(J\rightarrow 0^+)=+\infty \\
                        u_e(J\rightarrow +\infty)=+\infty
                    \end{aligned}
                    \right\}
                    \text{Condição de Crescimento}
                    \) 
            \end{itemize}
            \item \textbf{Modelo Constitutivo de Saint-Venant-Kirchhoff:}
            \begin{itemize}
                \item \(u_e = \frac{1}{2} E : \mathbb{C} : E  \quad \Rightarrow \quad \mathbb{C} = \frac{\partial^2 u_e}{\partial E \otimes \partial E}\)
                \item \(S=\frac{\partial u_e}{\partial E} \quad \Rightarrow \quad S = E : \mathbb{C}\) 
            \end{itemize}
        \end{itemize}
\end{frame}

\subsection{Determinação da Hessiana}
\begin{frame}
    \frametitle{Desenvolvimento Teórico}
    \framesubtitle{Determinação da Hessiana}
    \begin{itemize}
        \item \textbf{Hessiana:} Também denominada por \textit{Matriz de Rigidez Tangente}, é definida como:
    \end{itemize}



    \begin{equation}
        \frac{\partial^2 U}{ \partial Y_\gamma^z   \partial Y_\alpha^\beta} 
        = 
        \frac{\partial \left(F_\alpha^\beta\right)^{int}}{\partial Y_\gamma^z}
        =
        \int\limits_{V^{el}} \frac{\partial}{\partial Y_\gamma^z}\left(\frac{\partial \Psi(Y_k^m)}{\partial E_{km}} \frac{\partial E_{km}}{\partial Y_\gamma^\beta} \right) dV^{el}_0
    \end{equation}

    \begin{equation}
        H_{\alpha\beta\gamma z}^{el}
        = 
        \sum_{ih=1}^{nh} \underbrace{\left[h_{\alpha\beta\gamma z}\left(\vec{\xi(ih)}\right)\right] \vec{\xi}(ih)\omega_{ih}}_{\text{Domínio }\vec{\xi}} \underbrace{J_0 \left(\vec{\xi}(ih)\right)}_{\det \left[J\right]_0 = \det \mathbf{A}^0}
    \end{equation}

    \begin{itemize}
        \item Onde:
    \end{itemize}
    
    \begin{equation}
        h_{\alpha\beta\gamma z} = 
        \frac{\partial}{\partial Y_\gamma^z}\left(\frac{\partial \Psi(Y_k^m)}{\partial E_{km}} \frac{\partial E_{km}}{\partial Y_\gamma^\beta} \right)
    \end{equation}

\end{frame}


\begin{frame}
    \frametitle{Desenvolvimento Teórico}
    \framesubtitle{Determinação da Hessiana}

    \begin{equation}
        h_{\alpha\beta\gamma z} =
        \frac{\partial}{\partial Y_\gamma^z} 
        \underbrace{\left(\frac{\partial \Psi}{\partial E} :\frac{\partial E}{\partial Y_\gamma^\beta}\right) }_{\left(F_\gamma^\beta\right)^{int}}
    \end{equation}

    \begin{equation}
        h_{\alpha\beta\gamma z} =
        \frac{\partial E}{\partial Y_\gamma^\beta} :\underbrace{\frac{\partial^2 \Psi}{\partial E \otimes \partial E}}_{\mathbb{C} } : \frac{\partial E}{\partial Y_\alpha^\beta}
        +
        \underbrace{\frac{\partial \Psi}{\partial E }}_{\mathbf{S} } :\frac{\partial^2 E}{\partial Y_\gamma^z \partial Y^\beta_\alpha}
    \end{equation}

    \begin{equation}
        h_{\alpha\beta\gamma z} =
        \frac{\partial E}{\partial Y_\gamma^z} :\underbrace{\mathbb{C} : \frac{\partial E}{\partial Y_\alpha^\beta}}_{\frac{\partial S}{\partial Y_\alpha^\beta}}
        +
        \mathbf{S} :\frac{\partial^2 E}{\partial Y_\gamma^z \partial Y^\beta_\alpha}
    \end{equation}

    \begin{equation}
        h_{\alpha\beta\gamma z} =
        \frac{\partial E}{\partial Y_\gamma^z} :\frac{\partial S}{\partial Y_\alpha^\beta}
        +
        \mathbf{S} :\frac{\partial^2 E}{\partial Y_\gamma^z \partial Y^\beta_\alpha}
    \end{equation}

\end{frame}

\begin{frame}
    \frametitle{Desenvolvimento Teórico}
    \framesubtitle{Determinação da Hessiana}

    \begin{itemize}
        \item Desenvolvendo os termos \(\frac{\partial S}{\partial Y_\alpha^\beta}\) e \(\frac{\partial^2 E}{\partial Y_\gamma^z \partial Y^\beta_\alpha}\):
    \end{itemize}

    \begin{equation}
        DS_{\alpha\beta} = \frac{\partial S}{\partial Y_\alpha^\beta} = \mathbb{C} : \underbrace{\frac{\partial E}{\partial Y_\alpha^\beta}}_{DE_{\alpha\beta} }  
    \end{equation}


    \begin{equation}
        D2E_{\alpha\beta\gamma z} = \frac{\partial^2 E}{\partial Y_\gamma^z \partial Y^\beta_\alpha}  
    \end{equation}

    \begin{equation}
        \scalebox{0.8}{$
        D2E_{\alpha\beta\gamma z}
        =
        \frac{1}{2} \left[
        \left(A^0\right)^{-t} \left(DA^1_{\alpha\beta}\right)^t \left(DA^1_{\gamma z}\right)  \left(A^0\right)^{-1 }
        +\left(A^0\right)^{-t} \left(DA^1_{\gamma z}\right)\left(DA^1_{\alpha\beta}\right)^t  \left(A^0\right)^{-1}
        \right]
        $}
    \end{equation}

    \begin{equation}
        h_{\alpha\beta\gamma z} = DE_{\gamma z} : DS_{\alpha\beta} + S : D2E_{\alpha\beta\gamma z}
    \end{equation}

\end{frame}



\begin{frame}
    \frametitle{Desenvolvimento Teórico}
    \framesubtitle{Técnica de resolução}

    \begin{equation}
        \underbrace{\vec{g}}_{\text{Desbalanceamento Mecânico}} = \vec{F}_{\text{int}} - \vec{F}_{\text{ext}} 
        \quad \Rightarrow \quad
        \vec{g} = \dfrac{\partial \varPi}{\partial \vec{Y}}
    \end{equation}

    \begin{itemize}
        \item Expansão em série de Taylor ao redor de \(\vec{Y}_0\) (Posição tentativa):
    \end{itemize}

    \begin{equation}
        \vec{g}\underbrace{\left(\vec{Y}\right)}_{\text{Solução}} = \vec{g}\underbrace{\left(\vec{Y}_0\right)}_{\text{Tentativa}} + \underbrace{\left.\dfrac{\partial \vec{g}}{\partial \vec{Y}}\right|_{\vec{Y}_0}}_{\left.\frac{\partial^2\varPi }{\partial \vec{Y}^2}\right|_{\vec{Y}_0} = \mathbf{H}} \Delta \vec{Y} + \underbrace{\left.\dfrac{1}{2}\dfrac{\partial^2 \vec{g}}{\partial \vec{Y}^2}\right|_{\vec{Y_0}}\Delta \vec{Y}^2}_{\approx \vec{0}}
    \end{equation}

    \begin{equation}
        \vec{g}\left(\vec{Y}\right)\approx \vec{0} 
        \quad \Rightarrow \quad
        \underbrace{\vec{g}\left(\vec{Y}_0\right) + \left.\mathbf{H}\right|_{\vec{Y}_0}\Delta \vec{Y} = \vec{0}}_{\text{Sistema Linear}}
    \end{equation}

    \begin{equation}
        \Delta \vec{Y} = -\left[\left.\mathbf{H}\right|_{\vec{Y}_0}\right]^{-1}\vec{g}\left(\vec{Y}_0\right)  
    \end{equation}




\end{frame}





\begin{frame}

    \begin{algorithm}[H]
        \caption{Pseudo-Algoritmo}
        \begin{algorithmic}[1]
            \footnotesize % Adjust font size here
            \State{Primeira solução tentativa \( \vec{Y}^0 = \vec{X} \)}
            \For{Load Steps}
                \State{$\vec{F}^{ext} \gets \vec{F}^{ext} + \vec{\Delta F}^{ext}$ (\scriptsize{ Carregamento prescrito}) $\vec{Y} \gets \vec{Y} + \vec{\Delta Y}$ (\scriptsize{Posição prescrita})}
                \While{$\mathrm{RES} > \mathrm{TOL}$}
                    \For{each Element}
                        \State Compute $\left[\vec{(F^\beta_\alpha)}_{\text{int}}\right]^{el}$ \quad e \quad $\left[\mathbf{H} _{\alpha\beta\gamma z}\right]_{el}$
                    \EndFor

                    \State $\left(\vec{F}_{int}\right)^{\text{global}} \gets \left(\vec{F}_{int}\right)^{\text{global}} + \left(\vec{F}_{int}\right)^{\text{el}}$ \quad  e \quad  $\mathbf{H}^{\text{global}} \gets \mathbf{H}^{\text{global}} + \mathbf{H}^{\text{el}}$
                    \State $\vec{g} \gets  \vec{F}_{\text{int}} - \vec{F}_{\text{ext}}$
                    \State Condições de contorno sobre $\mathbf{H}$ e $\vec{g}$ 
                    \State $\mathbf{H} \cdot \vec{\Delta Y} = -\vec{g}$
                    \State $\vec{Y} \gets \vec{Y} + \vec{\Delta Y}$
                    \State $\mathrm{RES} = \dfrac{|\vec{\Delta Y}|}{|\vec{X}|}$
                \EndWhile
            \EndFor
        \end{algorithmic}
    \end{algorithm}
        

\end{frame}


\subsection{Formulação Dinâmica}
\begin{frame}
    \frametitle{Desenvolvimento Teórico}
    \framesubtitle{Formulação Dinâmica}

    \begin{itemize}
        \item Consideração de \(\mathbb{K} \quad \Rightarrow \quad \varPi = \mathbb{U} + \mathbb{P} + \mathbb{K}\)
    \end{itemize}

    \begin{equation}
        \mathbb{K} = \frac{1}{2} \int\limits_{V_0} \left(\rho^0 \dot{y_i} \dot{y_i} \right) dV^0
        \quad \Rightarrow \quad
        \delta \mathbb{K} = \frac{\delta \mathbb{K}}{dt}\delta t = \overbrace{\int\limits_{V_0} \left(\rho^0 \ddot{y_i} \delta y_i \right) dV^0}^{\vec{F}^{inercial}}
    \end{equation}

    \begin{equation}
        \delta \mathbb{K} = \int\limits_{V_0} \left(\rho^0 \phi_l \phi_\alpha \ddot{Y}_i^\alpha \right) dV^0 \delta Y_i^\ell
    \end{equation}

    \begin{itemize}
        \item Equilíbrio : \(\delta \varPi = \frac{\partial \varPi}{\partial Y_i^\ell}\delta Y_i^\ell \)
        \item Arbitrariedade de \(\delta Y_i^\ell \quad \Rightarrow \quad \delta \varPi = \frac{\partial \varPi}{\partial Y_i^\ell} = \frac{\partial \mathbb{U}}{\partial Y_i^\ell} + \frac{\partial \mathbb{P}}{\partial Y_i^\ell}  + \frac{\partial \mathbb{K}}{\partial Y_i^\ell} = 0 \)
    \end{itemize}

    \begin{equation}
        \frac{\partial \mathbb{K}}{\partial Y_i^\ell} = \int\limits_{V_0} \left(\rho^0 \phi_l \phi_\alpha \ddot{Y}_i^\alpha \right) dV^0 
        =
        \mathbf{M} \vec{\ddot{Y}} 
    \end{equation}

\end{frame}



\begin{frame}
    \frametitle{Desenvolvimento Teórico}
    \framesubtitle{Formulação Dinâmica}

    \begin{itemize}
        \item \textbf{Técnica de solução:} \textit{Newton-Raphson + Newmark}
    \end{itemize}

    \begin{equation}
        \scalebox{0.95}{$
        \vec{g} = \vec{F}_{\text{int}}\left(\vec{Y}\right) + \mathbf{M} \vec{\ddot{Y}} - \vec{F}_{\text{ext}}\left(\vec{Y}\right) = 0
        \quad\text{,}\quad 
        \mathbf{H}^{din} = \mathbf{H}^{est} + \dfrac{\mathbf{M}}{\gamma \Delta t^2} + \dfrac{\mathbf{C}}{\beta \Delta t}  
        $}
    \end{equation}

    
    \begin{itemize}
        \item Consideração de amortecimento \(\mathbf{C}\) proporcional a matriz de massa \(\mathbf{M}\)
        \item Integrador temporal de Newmark:
    \end{itemize}

    \begin{equation*}
        \vec{Y}^{t+1} = \vec{Y}^{t} + \Delta t \vec{\dot{Y}}^{t} + \left[\left(\dfrac{1}{2}-\beta\right)\vec{\ddot{Y}}^t + \beta \vec{\ddot{Y}}^{t+1} \right] \Delta t^2
    \end{equation*}
    \begin{equation*}
        \vec{\ddot{Y}}^{t+1} = \dfrac{\vec{Y}^{t+1}}{\beta \Delta t^2} - \vec{Q}^{t} 
        \quad \text{e} \quad  
        \vec{\dot{Y}}^{t+1} = \dfrac{\gamma}{\beta \Delta t} \vec{Y}^{t+1} + \vec{R}^{t} - \gamma \Delta t \vec{Q}^{t}
    \end{equation*}

    \begin{equation*}
        \vec{Q}^{t} = \frac{\vec{Y}^t}{\beta \Delta t^2} + \frac{\vec{\dot{Y}}^t}{\beta \Delta t} + \left( \frac{1}{2\beta} - 1 \right) \vec{\ddot{Y}}^t \quad \text{e} \quad \vec{R}^{t} = \left(\vec{\dot{Y}}^t + \left(1-\gamma\right)\Delta t \vec{\ddot{Y}}^t\right)
    \end{equation*}
\end{frame}


\begin{frame}

    \begin{algorithm}[H]
        \caption{Pseudo-Algoritmo}
        \begin{algorithmic}[1]
            \footnotesize % Adjust font size here
            \State{Primeira solução tentativa \( \vec{Y}^0 = \vec{X} \quad , \quad  \vec{\dot{Y}}^0=\vec{0} \quad , \quad \vec{\ddot{Y}}^0=\vec{0}\)}
            \For{Time Steps}
                \State{$\vec{F}^{ext} \gets \vec{F}^{ext} + \vec{\Delta F}^{ext}$ (\scriptsize{ Carregamento prescrito}) $\vec{Y} \gets \vec{Y} + \vec{\Delta Y}$ (\scriptsize{Posição prescrita})}
                \State{$\vec{Q}$ e $\vec{R}$}
                \While{$\mathrm{RES} > \mathrm{TOL}$}
                    \For{each Element}
                        \State Compute $\left[\vec{(F^\beta_\alpha)}_{\text{int}}\right]^{el}$ \quad e \quad $\left[\mathbf{H} _{\alpha\beta\gamma z}\right]_{el}$
                    \EndFor
                    \State $\left(\vec{F}_{int}\right)^{\text{global}} \gets \left(\vec{F}_{int}\right)^{\text{global}} + \left(\vec{F}_{int}\right)^{\text{el}}$ \quad  e \quad  $\mathbf{H}^{\text{global}} \gets \mathbf{H}^{\text{global}} + \mathbf{H}^{\text{el}}$
                    \State $\vec{g} \gets  \vec{F}_{\text{int}} - \vec{F}_{\text{ext}}$
                    \State Condições de contorno sobre $\mathbf{H}$ e $\vec{g}$ 
                    \State $\mathbf{H} \cdot \vec{\Delta Y} = -\vec{g}$
                    \State $\vec{Y} \gets \vec{Y} + \vec{\Delta Y}$
                    \State Atualizar: $\vec{\dot{Y}} \quad , \quad \vec{\ddot{Y}}$
                    \State $\mathrm{RES} = \dfrac{|\vec{\Delta Y}|}{|\vec{X}|}$
                \EndWhile
            \EndFor
        \end{algorithmic}
    \end{algorithm}
        

\end{frame}

\section{Modelo de Dano - Phase-Field}
\subsection{Formulação}

\begin{frame}
    \frametitle{Modelo de Dano}
    \framesubtitle{Phase-Field}
    \begin{itemize}
        \item \textbf{Funcional de \Textcite{Griffith1921}}: 
    \end{itemize}

    \begin{equation}
        \scalebox{0.9}{$
        \varPi(\varepsilon,\varGamma) = \int\limits_{\Omega} \psi_0 dV + \underbrace{G_c\int\limits_{\varGamma}  dS}_{\text{Dissipação na fratura}}  -  \Pi_{ext}
        $}
    \end{equation}


    \begin{columns}
        \column{0.3\textwidth}
        \begin{itemize}
            \item Regularização:
        \end{itemize}

        \column{0.7\textwidth}
        \begin{figure}
            \centering
            \begin{minipage}{0.4\textwidth}
                \centering
                \includegraphics[width=\linewidth]{figs/phaseFieldParameterD.png}
                \caption{Variável d}
            \end{minipage}
            \hfill
            \begin{minipage}{0.4\textwidth}
                \centering
                \includegraphics[width=\linewidth]{figs/regularização.png}
                \caption{Regularização}
            \end{minipage}
        \end{figure}
    \end{columns}

    \begin{equation}
        \scalebox{0.85}{$
        \varPi(\varepsilon,\varGamma) = \underbrace{\int\limits_{\Omega} \omega(d)\psi_0\left(\varepsilon\right) d\Omega}_{\text{Interna}} 
                                    + \underbrace{G_c\int\limits_{\Omega} \left(w(d) + c_d|\nabla d|^2\right) d\Omega}_{\text{Dissipação}} 
                                    - \underbrace{\Pi_{ext}}_{\text{Potêncial Externo}}
        $}
    \end{equation}
\end{frame}



\begin{frame}
    \frametitle{Modelo de Dano}
    \framesubtitle{Phase-Field}

    \begin{itemize}
        \item \textbf{Função degradação da energia interna:} \(\omega = \left(1 - d\right)^2\)
        \item \textbf{Funções definidoras da energia dissipada:}
    \end{itemize}

    \begin{columns}
        \column{0.5\textwidth}
        \[
            w(d)
            \left\{
            \begin{aligned}
                \frac{3}{8\ell_0}d \quad \text{Para o modelo AT1} \\
                \frac{1}{2\ell_0}d^2 \quad \text{Para o modelo AT2}
            \end{aligned}
            \right.
        \]
        \column{0.5\textwidth}
        \[
            c_d
            \left\{
            \begin{aligned}
                \frac{3}{8}\ell_0 \quad \text{Para o modelo AT1} \\
                \frac{1}{2}\ell_0 \quad \text{Para o modelo AT2}
            \end{aligned}
            \right.
        \]
    \end{columns}


\end{frame}


\begin{frame}
    \frametitle{Modelo de Dano}
    \framesubtitle{Phase-Field}

    \begin{itemize}
        \item \textbf{Degradação assimétrica da energia:} 
    \end{itemize}

    \begin{equation}
        \psi \left(\varepsilon, d\right) = \left(1 - d\right)^2\psi_0^+\left(\varepsilon\right)+\psi_0^-\left(\varepsilon\right)
    \end{equation} 

    \begin{equation}
        \sigma_{ij} = \frac{\partial \psi}{\partial \varepsilon_{ij}} = \left(1 - d\right)^2\left(2\mu \varepsilon_{ij}^D+\kappa\varepsilon_{ij}^{\upsilon+}\mathbf{I} \right) + \kappa \varepsilon_{ij}^{\upsilon-}\mathbf{I}
    \end{equation}


\end{frame}



\begin{frame}
    \frametitle{Modelo de Dano}
    \framesubtitle{Phase-Field}

        
    \begin{itemize}
        \item \textbf{Minimização do Funcional com relação a \(\vec{u}\)}: 
    \end{itemize}

    \begin{equation}
        \delta_{\boldsymbol{u}} \varPi = \int\limits_{\Omega} \partial_{\boldsymbol{u}} \Psi\left(\varepsilon, d\right) \left[\delta \boldsymbol{u}\right]d\Omega - \int\limits_{\Omega} \left(\boldsymbol{f} \cdot \delta_{\boldsymbol{u}}\right) d\Omega - \boldsymbol{F} \cdot \delta_{\boldsymbol{u}} \quad.
    \end{equation}


    \begin{itemize}
        \item \textbf{Minimização do Funcional com relação a \(d\)}: 
    \end{itemize}

    \begin{equation}
        \delta_d \varPi = \int\limits_{\Omega} \partial_d \Psi\left(\varepsilon, d\right) \left[\delta d\right]d\Omega + \frac{G_c}{l_0}\int\limits_{\Omega} d\left(\delta d\right)+l^2_0 \nabla d \cdot \nabla \left(\delta d\right) d\Omega
        \label{FunctionalVariation(d)}
    \end{equation}

    \begin{equation}
        \underbrace{\delta_d \varPi \geqslant 0 \quad \forall \delta d \geqslant 0 \quad \text{onde} \quad \delta d = d^{t_i+1} - d^{t_i}}_{\text{Irreversibilidade}} .
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Modelo de Dano}
    \framesubtitle{Phase-Field}

    \begin{itemize}
        \item \textbf{Escrevendo em forma incremental :}
        \begin{itemize}
            \item Expansão em série de Taylor em torno de \(d^{t_i}\) 
        \end{itemize}
    \end{itemize}

    \begin{equation}
        \scalebox{0.75}{$
        \varPi (u_{t_i},d_{t_i}+\Delta d_{t_i+1}) = \varPi(u_{t_i},d_{t_i}) + \underbrace{\nabla_d \varPi |_{d_{t_i}} \Delta d_{t_i+1} + \Delta d^T_{t_i+1} \left[\frac{1}{2} \left(\nabla^2_{dd} \varPi\right)|_{d_{t_i}}\right] \Delta d_{t_i+1} }_{\text{Parcela a ser minimizada}}
        $}
    \end{equation}
    \begin{itemize}
        \item \textbf{Definindo:}:
        \begin{itemize}
            \item \(\mathbf{Q} = \nabla^2_{dd} \varPi\left(u_{t_i},d_n\right)\)
            \item \(\vec{q} = \nabla_d \varPi\left(u_{t_i},d_n\right)\)
        \end{itemize}
        \item Reecreve-se \(\varPi\) do modo como \textcite{Marengo2021}
    \end{itemize}

    \begin{equation}
        \varPi_{t_i} = \frac{1}{2} \Delta d^T_{t_i+1} \mathbf{Q} \Delta d_{t_i+1} + \vec{q} \Delta d_{t_i+1} + \varPi_{t_i}
    \end{equation}

\end{frame}

\begin{frame}
    \frametitle{Modelo de Dano}
    \framesubtitle{Phase-Field}

    \begin{itemize}
        \item \textbf{Definição de \(\mathbf{Q}\) e \(\vec{q}\)}:
    \end{itemize}

    \begin{equation}
        \mathbf{Q} = \varPsi(u_{t_i}) + G_c \varPhi \qquad \text{e} \qquad \vec{q} = \mathbf{Q}\vec{d_{t_i}} - \boldsymbol{\varphi}(u_{t_i})
    \end{equation}


    \begin{equation}
        \varPhi = \int\limits_{\Omega} \left(\ell_0^{-1}\mathbf{N}^T \mathbf{N} + \ell_0\mathbf{B}^T \mathbf{B}\right) d\Omega \quad \text{onde} \quad \mathbf{B} = \mathbf{J}_0^{-1}\frac{\partial \vec{\phi}}{\partial\vec{\xi }}
    \end{equation}

    \begin{equation}
        \varPsi = \int\limits_{\Omega} \left(2\psi_0^+u_{t_i}\mathbf{N}^T \mathbf{N}\right) d\Omega
    \end{equation}

    \begin{equation}
        \boldsymbol{\varphi} = \int\limits_{\Omega} \left(\psi_0^+u_{t_i}\mathbf{N}^T \right) d\Omega
    \end{equation}


\end{frame}


\begin{frame}
    \frametitle{Modelo de Dano}
    \framesubtitle{Phase-Field}

    \begin{itemize}
        \item Finalmente, a solução do problema incremental se dá pela minimização \textbf{restrita} :
    \end{itemize}

    \begin{equation}
        \Delta \boldsymbol{d} = \text{argmin}\left\{\frac{1}{2}\Delta \boldsymbol{d}^T \mathbf{Q} \Delta \boldsymbol{d} + \vec{q} \Delta \boldsymbol{d} \quad : \quad \Delta \boldsymbol{d} \geqslant 0 \right\}
    \end{equation}

    \begin{itemize}
        \item Que pode ser reescrito :
    \end{itemize}

    \begin{equation}
        \left(\mathbf{Q}\Delta\vec{d}+\vec{q}\right)^T\Delta\vec{d} = 0 \quad \text{e} \quad -\left(\mathbf{Q}\Delta \vec{d} +\vec{q}\right) \leqslant 0 \quad \text{e} \quad \Delta \vec{d} \geqslant 0
    \end{equation}

    \begin{itemize}
        \item O problema é resolvido pelo algoritmo PSOR \Textcite{Marengo2021}
    \end{itemize}

\end{frame}

% Include in your preamble


\begin{frame}

    \begin{algorithm}[H]
        \caption{Pseudo-Algoritmo}
        \begin{algorithmic}[1]
            \footnotesize % Adjust font size here
            \State{Primeira solução tentativa \( \vec{Y}^0 = \vec{X},\ \vec{d}^0=\vec{0}\)}
            \For{Load Steps}
                \State{$\vec{F}^{ext} \gets \vec{F}^{ext} + \Delta \vec{F}^{ext}$} \Comment{Carregamento prescrito}
                \State{$\vec{Y} \gets \vec{Y} + \Delta \vec{Y}$} \Comment{Posição prescrita}
                \While{$\mathrm{RES} > \mathrm{TOL}$}
                    \State{$\delta_{\boldsymbol{u}}\, \varPi\left(u^i , d^i\right) = 0 \quad \rightarrow \quad u^i$}
                    \State{$\delta_d\, \varPi\left(u^i , d^n\right) \geq 0  \quad \rightarrow \quad d^i$}
                    \State{$\mathrm{RES} = \left| \vec{F}^{int}\left(u^i , d^i\right) \right|$}
                \EndWhile
                \State{$\left(u^{n+1} , d^{n+1} \right) \gets \left(u^{i} , d^{i} \right)$}
            \EndFor
        \end{algorithmic}
    \end{algorithm}

\end{frame}
